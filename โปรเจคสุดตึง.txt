local function GetBouncingColor()
    local sinValue = math.sin(tick() * 2)
    local lerpFactor = (sinValue + 1) / 2
    return Color3.fromRGB(0, 0, 0):Lerp(Color3.fromRGB(255, 0, 0), lerpFactor)
end

local function Tween(obj, info, goal)
    return TweenService:Create(obj, TweenInfo.new(info), goal):Play()
end

local function makeDraggable(topbar, object)
	local dragging, dragInput, dragStart, startPos
	topbar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = object.Position
		end
	end)
	topbar.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	UIS.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			object.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
	topbar.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

-- [[ GUN MOD LOGIC ]] --
local function ApplyGunMod(mode)
    local char = LocalPlayer.Character
    local tool = char and char:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("Setting") then
        local gun = require(tool.Setting)
        gun.LimitedAmmoEnabled = false ; gun.Ammo = math.huge ; gun.MaxAmmo = math.huge ; gun.ReloadTime = 0 
        gun.Auto = true ; gun.Recoil = 0 ; gun.Accuracy = 1 ; gun.Spread = 0 ; gun.BulletSpeed = 1000

        if mode == "Freeze" then
            gun.FreezingBullet = true ; gun.IcifyChance = 100 ; gun.FlamingBullet = false
        elseif mode == "Fire" then
            gun.FreezingBullet = false ; gun.FlamingBullet = true ; gun.BurnChance = 100
        elseif mode == "Mixed" then
            gun.FreezingBullet = true ; gun.IcifyChance = 100 
            gun.FlamingBullet = true ; gun.BurnChance = 100
        elseif mode == "Normal" then
            gun.FreezingBullet = false ; gun.FlamingBullet = false
        end
        return true
    end
    return false
end

-- // FUNCTION: สร้างหรือดึง Label (Object Management)
local function GetOrCreateLabel(container, index, itemName, statusIcon)
    local labelName = "Item_" .. index
    local label = container:FindFirstChild(labelName)

    if not label then
        label = Instance.new("TextLabel")
        label.Name = labelName
        label.Size = UI_SETTINGS.LabelSize
        label.TextColor3 = Color3.new(1, 1, 1)
        label.Font = UI_SETTINGS.Font
        label.TextSize = UI_SETTINGS.TextSize
        label.ZIndex = 10
        label.Parent = container

        -- เพิ่มความโค้งมน
        local corner = Instance.new("UICorner", label)
        corner.CornerRadius = UDim.new(0, 6)

        -- เพิ่มเส้นขอบพรีเมียม
        local stroke = Instance.new("UIStroke", label)
        stroke.Thickness = 1.5
        stroke.Transparency = 0.5
        stroke.Color = Color3.new(1, 1, 1)

        -- เพิ่ม Gradient ให้ดูมีมิติ
        local gradient = Instance.new("UIGradient", label)
        gradient.Rotation = 90
        gradient.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.new(1, 1, 1)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(200, 200, 200))
        })
    end

    -- อัปเดตข้อมูล (Update Properties)
    local isEquipped = (statusIcon == "⚔️")
    label.Text = string.format("%s  %s", statusIcon, itemName:upper())
    label.BackgroundColor3 = isEquipped and UI_SETTINGS.EquippedColor or UI_SETTINGS.BackpackColor
    label.BackgroundTransparency = 0.2
    label.Visible = true
    
    return label
end

-- // MAIN LOOP
task.spawn(function()
    while true do
        if totalInvActive then
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
                    local head = p.Character.Head
                    local tag = head:FindFirstChild("TotalInvTag") or CreateTotalInvTag(head)
                    local container = tag.Container
                    
                    local currentItems = {}

                    -- 1. เก็บข้อมูลไอเทมที่ถืออยู่
                    for _, t in pairs(p.Character:GetChildren()) do
                        if t:IsA("Tool") and not UI_SETTINGS.ExcludedTools[t.Name] then
                            table.insert(currentItems, {name = t.Name, icon = "⚔️"})
                        end
                    end

                    -- 2. เก็บข้อมูลไอเทมในกระเป๋า
                    if p:FindFirstChild("Backpack") then
                        for _, t in pairs(p.Backpack:GetChildren()) do
                            if t:IsA("Tool") and not UI_SETTINGS.ExcludedTools[t.Name] then
                                table.insert(currentItems, {name = t.Name, icon = "🎒"})
                            end
                        end
                    end

                    -- 3. อัปเดต UI (Reuse Labels)
                    -- ซ่อน Label เก่าก่อน
                    for _, child in pairs(container:GetChildren()) do
                        if child:IsA("TextLabel") then child.Visible = false end
                    end

                    -- แสดงผลรายการใหม่
                    for i, item in ipairs(currentItems) do
                        GetOrCreateLabel(container, i, item.name, item.icon)
                    end
                end
            end
        end
        task.wait(0.5)
    end
end)

        if not char then return end

            h.Enabled = espActive
            if espActive then h.FillColor = GetBouncingColor() end 
        end)
    end
    targetPlayer.CharacterAdded:Connect(addESP)
    if targetPlayer.Character then addESP(targetPlayer.Character) end
end
